③ Apps Script 코드
function doGet(e) {
  var action = (e && e.parameter && e.parameter.action) || 'getReviews';
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  if (action === 'getReviews') {
    var sh = ss.getSheetByName('검토현황') || ss.insertSheet('검토현황');
    return sheetToJson(sh, 'reviews');
  }
  if (action === 'getComps') {
    var sh = ss.getSheetByName('타사레퍼런스');
    if (!sh) return out({type:'comps', rows:[]});
    return sheetToJson(sh, 'comps');
  }
  if (action === 'getCompRef') {
    var sh = ss.getSheetByName('경쟁사 참고');
    if (!sh) return out({type:'compRef', rows:[]});
    return sheetToJson(sh, 'compRef');
  }
  return out({error:'unknown action'});
}

function doPost(e) {
  var body = JSON.parse(e.postData.contents);
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  if (body.type === 'reviews') {
    var sh = ss.getSheetByName('검토현황') || ss.insertSheet('검토현황');
    var labels = ['국가','접점','문안번호','문안','리스크','검토의견','담당자','검토일'];
    var keys   = ['country','touchpoint','claimIdx','claim','risk','review','reviewer','date'];
    return writeSheet(sh, labels, keys, body.rows);
  }
  if (body.type === 'comps') {
    var sh = ss.getSheetByName('타사레퍼런스') || ss.insertSheet('타사레퍼런스');
    var labels = ['국가','접점','슬롯번호','브랜드','클레임문구'];
    var keys   = ['country','touchpoint','slotIdx','brand','copy'];
    return writeSheet(sh, labels, keys, body.rows);
  }
  return out({error:'unknown type'});
}

function sheetToJson(sh, type) {
  var data = sh.getDataRange().getValues();
  if (data.length <= 1) return out({type:type, rows:[]});
  var headers = data[0];
  var rows = data.slice(1).map(function(r) {
    var obj = {};
    headers.forEach(function(h,i){ obj[h] = r[i]!=null?String(r[i]):''; });
    return obj;
  });
  return out({type:type, rows:rows});
}

function writeSheet(sh, labels, keys, rows) {
  sh.clearContents();
  sh.getRange(1,1,1,labels.length).setValues([labels]);
  if (rows && rows.length) {
    var vals = rows.map(function(r){
      return keys.map(function(k){ return r[k]!=null?String(r[k]):''; });
    });
    sh.getRange(2,1,vals.length,keys.length).setValues(vals);
  }
  return out({status:'ok', count:(rows||[]).length});
}

function out(data) {
  return ContentService
    .createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

방법 A. GitHub Pages 무료 · 권장
1github.com 에서 새 리포지터리 생성 (예: anua-derma)
2이 HTML 파일을 index.html 로 이름 변경 후 업로드
3Settings → Pages → Branch: main → Save
4https://[계정명].github.io/anua-derma 를 팀에 공유
방법 B. Netlify Drop 계정 불필요 · 즉시
1app.netlify.com/drop 접속
2HTML 파일을 드래그앤드롭 → 즉시 공개 URL 발급
⚠️ 팀 공용 URL 설정법 (권장): Apps Script 배포 후 HTML 파일 소스에서 DEFAULT_APPS_SCRIPT_URL 상수 값을 웹앱 URL로 바꾸면, 외부 링크로 접속하는 모든 팀원이 별도 설정 없이 자동으로 Google Sheets와 연동됩니다.

또는 각 접속자가 ⚙ 연동 설정에서 동일한 URL을 직접 입력하는 방법도 가능합니다.
